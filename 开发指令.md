# 视频课堂质量分析看板系统 - 最终开发文档

## 📋 项目概述

开发一个视频课堂质量分析的 Web 应用系统，支持管理员和普通用户上传课堂视频 URL 或批量导入数据，后端调用听悟 ASR 和豆包大模型进行自动分析，最终在看板页面展示分析结果。

**核心功能：**
- 用户认证与权限管理（管理员/普通用户）
- 视频 URL 上传（单条/批量 Excel/CSV）
- 自动化视频处理流程（听悟 ASR → 下载视频 → 分帧 → 上传 OSS → 豆包分析 → 保存结果）
- 实时任务进度追踪
- 可视化看板展示分析结果
- 支持多维度筛选

### 步骤4：分帧

**分帧规则：**
- 10分钟：随机抽 1 张
- 20分钟：随机抽 1 张
- 20分钟后：每10分钟抽3张（间隔3分钟）
  - 例如30分钟：30分、33分、36分
- 结尾前5秒：抽 1 张

```python
def extract_frames_step(video_id):
    """分帧并保存到本地"""
    local_video_path = f"./videos/{video_id}.mp4"
    frames_dir = f"./frames/{video_id}/"
    
    update_step_status(video_id, "extract_frames", "processing")
    
    try:
        os.makedirs(frames_dir, exist_ok=True)
        
        # 按规则抽帧
        frame_paths = extract_frames_by_rules(local_video_path, frames_dir)
        
        update_step_status(video_id, "extract_frames", "completed")
        return frame_paths
    except Exception as e:
        handle_error(video_id, "extract_frames", e, max_retries=3)


def extract_frames_by_rules(video_path, output_dir):
    """按规则抽帧"""
    import cv2
    import random
    
    cap = cv2.VideoCapture(video_path)
    fps = cap.get(cv2.CAP_PROP_FPS)
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    duration_seconds = total_frames / fps
    
    frame_paths = []
    frame_count = 0
    
    def extract_frame_at_second(second):
        """在指定秒数提取帧"""
        nonlocal frame_count
        frame_number = int(second * fps)
        cap.set(cv2.CAP_PROP_POS_FRAMES, frame_number)
        ret, frame = cap.read()
        
        if ret:
            frame_count += 1
            frame_path = os.path.join(output_dir, f"frame_{frame_count:03d}.jpg")
            cv2.imwrite(frame_path, frame)
            frame_paths.append(frame_path)
    
    # 规则1：10分钟处随机抽1张
    if duration_seconds >= 600:
        random_second = random.randint(0, 600)
        extract_frame_at_second(random_second)
    
    # 规则2：20分钟处随机抽1张
    if duration_seconds >= 1200:
        random_second = random.randint(600, 1200)
        extract_frame_at_second(random_second)
    
    # 规则3：20分钟后，每10分钟抽3张（间隔3分钟）
    if duration_seconds > 1200:
        current_minute = 30
        while current_minute * 60 < duration_seconds - 5:
            for offset in [0, 3, 6]:
                second = (current_minute + offset) * 60
                if second < duration_seconds - 5:
                    extract_frame_at_second(second)
            current_minute += 10
    
    # 规则4：结尾前5秒抽1张
    if duration_seconds > 5:
        extract_frame_at_second(duration_seconds - 5)
    
    cap.release()
    return frame_paths
```

def save_results_step(video_id):
    """解析豆包结果并保存到数据库"""
    video = get_video_info(video_id)
    llm_result_json = video['llm_result_json']
    
    update_step_status(video_id, "save_results", "processing")
    
    try:
        result = json.loads(llm_result_json)
        evaluation_results = result["class_video_analysis"]["evaluation_results"]
        
        for category in evaluation_results:
            for item in category["items"]:
                behavior_code = item["behavior_code"]
                is_compliant = item["is_compliant"]
                evidence_timestamp = ",".join(item.get("evidence_timestamp", []))
                analysis_comment = item.get("analysis_comment", "")
                
                # 查询 category_id
                category_id = get_category_id_by_code(behavior_code)
                
                if category_id is None:
                    print(f"⚠ 找不到 behavior_code: {behavior_code}")
                    continue
                
                # 插入评估记录
                insert_evaluation(
                    video_id=video_id,
                    category_id=category_id,
                    is_compliant=is_compliant,
                    evidence_timestamp=evidence_timestamp,
                    analysis_comment=analysis_comment
                )
        
        update_step_status(video_id, "save_results", "completed")
        update_video_info(video_id, task_status="completed")
    except Exception as e:
        handle_error(video_id, "save_results", e, max_retries=0)

### 错误处理与重试

```python
def handle_error(video_id, step, exception, max_retries=3):
    """统一错误处理"""
    video = get_video_info(video_id)
    current_retry = video['retry_count']
    
    is_network_err = is_network_error(exception)
    
    if is_network_err and current_retry < max_retries:
        wait_time = 5 * (2 ** current_retry)  # 5s, 10s, 20s
        print(f"⚠ 网络错误，{wait_time}秒后重试（{current_retry+1}/{max_retries}）")
        time.sleep(wait_time)
        
        update_video_info(video_id, retry_count=current_retry + 1)
        execute_step(video_id, step)
    else:
        error_msg = f"{step} 步骤失败：{str(exception)}"
        update_video_info(
            video_id=video_id,
            task_status="failed",
            current_step=step,
            error_message=error_msg
        )
        update_step_status(video_id, step, "failed")
        raise exception


def is_network_error(exception):
    """判断是否为网络错误"""
    error_types = ["ConnectionError", "Timeout", "HTTPError", "URLError"]
    return any(err in str(type(exception).__name__) for err in error_types)
```

### 继续任务功能

```python
def retry_task(video_id):
    """从失败的步骤继续执行"""
    video = get_video_info(video_id)
    
    if video['task_status'] != 'failed':
        return {"success": False, "message": "任务未失败，无需继续"}
    
    # 重置状态
    update_video_info(
        video_id=video_id,
        task_status="processing",
        error_message=None,
        retry_count=0
    )
    
    # 从失败的步骤继续
    failed_step = video['current_step']
    execute_from_step(video_id, failed_step)
    
    return {
        "success": True,
        "message": f"任务已重新启动，从 {failed_step} 步骤继续"
    }


def execute_from_step(video_id, start_step):
    """从指定步骤开始执行"""
    steps = [
        "asr",
        "parse_asr",
        "download",
        "extract_frames",
        "upload_oss",
        "llm_analysis",
        "save_results"
    ]
    
    # 特殊处理：如果从 upload_oss 继续，检查是否有 frame_urls
    if start_step == "upload_oss":
        video = get_video_info(video_id)
        if video.get('frame_urls'):
            # 已有 OSS URL，跳过上传，从豆包开始
            start_step = "llm_analysis"
    
    start_index = steps.index(start_step)
    
    for step in steps[start_index:]:
        execute_step(video_id, step)
    
    # 清理图片
    cleanup_files_step(video_id)


def execute_step(video_id, step):
    """执行单个步骤"""
    step_functions = {
        "asr": asr_step,
        "parse_asr": parse_asr_step,
        "download": download_video_step,
        "extract_frames": lambda vid: (extract_frames_step(vid), delete_video_step(vid))[0],
        "upload_oss": upload_oss_step,
        "llm_analysis": llm_analysis_step,
        "save_results": save_results_step
    }
    
    if step in step_functions:
        step_functions[step](video_id)
```

## ✅ 关键要点总结

1. **流程顺序：** ASR → 解析 → 下载 → 分帧 → **删除视频** → 上传OSS → 豆包 → 保存 → 清理图片
2. **分帧规则：** 10分/20分各1张，20分后每10分3张（间隔3分钟），结尾前5秒1张
3. **数据保存：** 
   - ✅ 保存 OSS 图片 URL（逗号分隔）
   - ❌ 不保存分帧时间戳
   - ❌ 不保存本地路径（处理完清理）
4. **继续功能：** 从失败步骤继续，如果已有 frame_urls 则跳过上传OSS
5. **评估类别：** 只保存子类（10个），通过 parent_category 字段关联父类
6. **前端筛选：** 父类按字符串完全匹配，子类按ID匹配
7. **错误重试：** 网络错误重试3次（5s/10s/20s）
8. **外部脚本：** 你提供4个脚本（听悟ASR、ASR处理、OSS上传、豆包模型）
